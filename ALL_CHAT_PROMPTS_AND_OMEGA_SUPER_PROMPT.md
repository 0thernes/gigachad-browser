function tokenize(text) {
  return (text || '')
    .toLowerCase()
    .replace(/[^a-z0-9\s]/g, ' ')
    .split(/\s+/)
    .filter(w => w.length > 2);
}

function searchMemoryVault(query, topK = 3) {
  if (!memoryVault.length || !query) return [];
  const qTokens = new Set(tokenize(query));
  if (!qTokens.size) return [];

  const scored = memoryVault.map(item => {
    const docTokens = tokenize(item.title + ' ' + item.summary);
    let hits = 0;
    for (const t of docTokens) {
      if (qTokens.has(t)) hits++;
    }
    const score = hits / Math.sqrt(docTokens.length + 10);
    return { item, score };
  });

  return scored
    .filter(s => s.score > 0.05)
    .sort((a, b) => b.score - a.score)
    .slice(0, topK)
    .map(s => s.item);
}

function downloadDirectFile(url, destPath) {
  return new Promise((resolve, reject) => {
    const client = url.startsWith('https') ? https : http;
    client.get(url, { headers: { 'User-Agent': 'Mozilla/5.0' } }, (res) => {
      if (res.statusCode >= 300 && res.statusCode < 400 && res.headers.location) {
        return downloadDirectFile(res.headers.location, destPath).then(resolve).catch(reject);
      }
      const file = fs.createWriteStream(destPath);
      res.pipe(file);
      file.on('finish', () => file.close(() => resolve(destPath)));
    }).on('error', reject);
  });
}

function cosineSimilarity(a, b) {
  if (!a || !b || a.length !== b.length) return 0;
  let dot = 0, normA = 0, normB = 0;
  for (let i = 0; i < a.length; i++) {
    dot += a[i] * b[i];
    normA += a[i] * a[i];
    normB += b[i] * b[i];
  }
  return dot / (Math.sqrt(normA) * Math.sqrt(normB) + 1e-8);
}

function tokenize(text) {
  return (text || '').toLowerCase().replace(/[^a-z0-9\s]/g, ' ').split(/\s+/).filter(w => w.length > 2);
}

async function searchMemoryVault(query, ollamaBaseUrl, topK = 3) {
  if (!memoryVault.length || !query) return [];
  const qVec = await getGemma2Embedding(query, true, ollamaBaseUrl);
  const qTokens = new Set(tokenize(query));

  const scored = memoryVault.map(item => {
    let score = 0;
    if (qVec && item.embedding) {
      score = cosineSimilarity(qVec, item.embedding);
    } else {
      const docTokens = tokenize(item.title + ' ' + item.summary);
      let hits = 0;
      for (const t of docTokens) if (qTokens.has(t)) hits++;
      score = hits / Math.sqrt(docTokens.length + 10);
    }
    return { item, score };
  });

  return scored.filter(s => s.score > 0.08).sort((a, b) => b.score - a.score).slice(0, topK).map(s => s.item);
}

async function attachAndEnableCDP(wc) {
  const wcId = wc.id;
  if (!wc.debugger.isAttached()) wc.debugger.attach('1.3');
  if (!attachedDebuggers.has(wcId)) {
    attachedDebuggers.add(wcId);
    snifferBuffer.set(wcId, []);
    await wc.debugger.sendCommand('Network.enable', { maxPostDataSize: 65536 });

    wc.debugger.on('message', async (event, method, params) => {
      if (method === 'Network.responseReceived') {
        const mime = (params.response?.mimeType || '').toLowerCase();
        const url = params.response?.url || '';
        if (mime.includes('json') && !url.startsWith('data:')) {
          try {
            const bodyObj = await wc.debugger.sendCommand('Network.getResponseBody', { requestId: params.requestId });
            if (bodyObj?.body) {
              const list = snifferBuffer.get(wcId) || [];
              list.unshift({ url: url.slice(0, 140), status: params.response.status, preview: bodyObj.body.slice(0, 2500) });
              if (list.length > 15) list.pop();
              snifferBuffer.set(wcId, list);
            }
          } catch (_) {}
        }
      }
    });
    wc.on('destroyed', () => { attachedDebuggers.delete(wcId); snifferBuffer.delete(wcId); mediaStreamBuffer.delete(wcId); });
  }
}

      await new Promise(r => setTimeout(r, 45));

      return { ok: true, detail: `CDP Click @ (${x}, ${y})` };
    }
    if (cmd.action === 'type' && typeof cmd.x === 'number') {
      const x = Math.round(cmd.x), y = Math.round(cmd.y);
      await wc.debugger.sendCommand('Input.dispatchMouseEvent', { type: 'mousePressed', x, y, button: 'left', clickCount: 1 });
      await wc.debugger.sendCommand('Input.dispatchMouseEvent', { type: 'mouseReleased', x, y, button: 'left', clickCount: 1 });
      await new Promise(r => setTimeout(r, 60));
      if (cmd.text) await wc.debugger.sendCommand('Input.insertText', { text: cmd.text });
      if (cmd.pressEnter !== false) {
        await wc.debugger.sendCommand('Input.dispatchKeyEvent', { type: 'keyDown', windowsVirtualKeyCode: 13, key: 'Enter', code: 'Enter' });
        await wc.debugger.sendCommand('Input.dispatchKeyEvent', { type: 'keyUp', windowsVirtualKeyCode: 13, key: 'Enter', code: 'Enter' });
